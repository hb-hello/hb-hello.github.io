% Content macros
% scripts/sync-resume.js parses the document body into website entries.
% Inside \begin{document}, stick to these macros plus inline \textbf,
% \textit, \emph, \underline, \href, \resumeLink, \textbar{}, \%, \&, --, ~ and \\.
% Every entry gets an [id]; the website places entries by that id.
%
%   \begin{entries} ... \end{entries}                 list of entries in a section
%   \education[id]{school}{location}{degree}{dates}
%   \begin{job}[id]{title}{org}{location}{dates}      \resumeItem{...} ... \end{job}
%   \begin{project}[id]{name}{org}{dates}             \resumeItem{...} ... \end{project}
%   \begin{skills}[id] \skill{category}{items} ... \end{skills}
%
% Optional fields, mostly website-only. Put them inside a job or project
% right after its heading, before its \resumeItems:
%   \tech{languages}  \repo{url}  \notes{url}  \location{place} (projects only)
% Only \repo shows on the PDF: a project's heading gets " | Git Repo" linking
% to it. Write % in a url as \%; a \repo url can't contain #.
\newenvironment{entries}{\resumeSubHeadingListStart}{\resumeSubHeadingListEnd}
\newcommand{\education}[5][]{\resumeSubheading{#2}{#3}{#4}{#5}}
\newenvironment{job}[5][]{\resumeSubheading{#2}{#5}{#3}{#4}\resumeItemListStart}{\resumeItemListEnd}
\newenvironment{skills}[1][]{\begin{itemize}[leftmargin=0.15in, label={}]\small\begingroup\item\begingroup\gdef\skillsep{}}{\endgroup\endgroup\end{itemize}}
\newcommand{\skill}[2]{\skillsep\textbf{#1}{: #2}\gdef\skillsep{\\}}

\definecolor{linkblue}{rgb}{0.0, 0.27, 0.65}
% Blue link text with a thin underline close to the baseline:
%   \resumeLink[size]{url}{text}   size is optional, e.g. \footnotesize
\newcommand{\resumeLink}[3][]{\href{#2}{#1\color{linkblue}%
  \setlength{\ULdepth}{1.2pt}\renewcommand{\ULthickness}{0.4pt}\uline{#3}}}

\makeatletter
% The fields print nothing where they appear. Each body is different so the
% project scanner below can tell them apart with \@ifnextchar.
\newcommand{\tech}[1]{\@gobble{tech}}
\newcommand{\repo}[1]{\@gobble{repo}}
\newcommand{\notes}[1]{\@gobble{notes}}
\newcommand{\location}[1]{\@gobble{location}}

% \begin{project} reads the field commands that follow its heading arguments,
% keeping \repo's url, then typesets the heading.
\newenvironment{project}[4][]{%
  \def\project@args{{#2}{#3}{#4}}\def\project@repo{}\project@scan
}{\resumeItemListEnd}
\def\project@scan{%
  \@ifnextchar\repo{\project@takerepo}{%
  \@ifnextchar\tech{\project@skip}{%
  \@ifnextchar\notes{\project@skip}{%
  \@ifnextchar\location{\project@skip}{%
  \expandafter\project@heading\project@args}}}}}
\def\project@takerepo\repo#1{\def\project@repo{#1}\project@scan}
\def\project@skip#1#2{\project@scan}
\def\project@heading#1#2#3{%
  \resumeProjectHeading{\textbf{#1} \textbar{} \emph{#2}%
    \ifx\project@repo\@empty\else{} \textbar{} \resumeLink[\footnotesize]{\project@repo}{Github repo}\fi}{#3}%
  \resumeItemListStart}
\makeatother
